COMMANDS FOR OPERATOR B (PowerShell, normal window, NOT as administrator). The same numbers as in README_FOR_OPERATOR_B.md.
Run every block from the folder C:\slwlab-b (the unzipped kit). Copy a block, paste it, read the output, go on only if it says what the "Expect" line says.
If something else appears: STOP, copy the text of the screen and send it to the organiser. Never send the files .env or secrets\*.

=== STEP 2: unzip and check the files ===
New-Item -ItemType Directory -Force C:\slwlab-b | Out-Null
Expand-Archive -Path "$env:USERPROFILE\Downloads\slwlab-kit-b.zip" -DestinationPath C:\slwlab-b -Force
cd C:\slwlab-b
Get-Content CHECKSUMS.txt | ForEach-Object { $h, $f = $_ -split '\s+', 2; $f = $f.Trim(); "{0}  {1}" -f ((Get-FileHash $f -Algorithm SHA256).Hash.ToLower() -eq $h), $f }
# Expect: every line starts with True. The sender.exe line must say True too (475fe47a880a...).
Test-Path secrets\master.json
# Expect: True (the organiser delivers it in the kit or as a separate file: put it in C:\slwlab-b\secrets\).

=== STEP 3: Visual C++ runtime ===
Test-Path C:\Windows\System32\vcruntime140.dll
.\sender.exe --help | Select-Object -First 3
# Expect: True, then "Leader-window ordering experiment". If the program does not start or a window says vcruntime140.dll is missing:
# download https://aka.ms/vs/17/release/vc_redist.x64.exe (Microsoft), run it, then repeat these two lines.

=== STEP 4: Helius keys into .env ===
cd C:\slwlab-b
Copy-Item env.template .env
notepad .env
# Paste RPC_URL and RPC_WS_URL from your Helius dashboard after the "=" (no spaces, no quotes). Save, close Notepad.
(Select-String -Path .env -Pattern '^(RPC_URL|RPC_WS_URL)=.+api-key=' ).Count
# Expect: 2.

=== STEP 5: laptop settings (charger plugged in the whole time) ===
powercfg /change standby-timeout-ac 0
powercfg /change hibernate-timeout-ac 0
powercfg /setacvalueindex SCHEME_CURRENT 4f971e89-eebd-4455-a8de-9e59040e7347 5ca83367-6e45-459f-a27b-476b1d01c936 0
powercfg /setacvalueindex SCHEME_CURRENT 19cbb8fa-5279-450e-9fac-8a3d5fedd0c1 12bbebe6-58d6-4636-95bb-3217ef867c1a 0
powercfg /setactive SCHEME_CURRENT
Get-CimInstance Win32_Battery | Select-Object BatteryStatus
Get-PSDrive C | Select-Object @{n='FreeGB';e={[math]::Round($_.Free/1GB,1)}}
# Expect: BatteryStatus 2 (on AC power), FreeGB at least 10. The first three lines: never sleep on the charger, the lid does nothing; the fourth: Wi-Fi at full performance.
# Also by hand: Settings > Windows Update > Pause updates; close other programs; connect to the usual Wi-Fi; do not use a VPN; do not close the lid.

=== STEP 6: keys of the payers, schedule, first balances (nothing is sent) ===
cd C:\slwlab-b
.\sender.exe init-payers --count 32
.\sender.exe leader-snapshots
.\sender.exe balances
# Expect: 32 lines payer_000 ... payer_031, a line "sink ...", a line "master ... (existing)"; then a balances file; "total" about 30000000 lamports.
# STOP: "master ... (new; fund this wallet)" - this means secrets\master.json is missing or in the wrong folder. Tell the organiser.

=== STEP 7: fund the payers (dry run first) ===
cd C:\slwlab-b
.\sender.exe fund --payer-target-lamports 771101
# Expect: ends with "dry-run: nothing sent"; 33 transfers; the sum about 25.3 million lamports. STOP if the sum is above 26500000.
$env:CONFIRM_LIVE = 'yes'
.\sender.exe fund --payer-target-lamports 771101 --live --max-spend-lamports 200000
Remove-Item Env:CONFIRM_LIVE
$POST = (Get-ChildItem data\snapshots\balances_*.json | Sort-Object Name | Select-Object -Last 1).FullName
$POST
# Expect: "ok": true, "problems": []. If some transfers expired, run the dry run and the live line again (it is safe to repeat).
# Keep this PowerShell window open (the variable $POST is used in step 11), or write the file name down.

=== STEP 8: smoke session, 20 minutes ===
$env:CONFIRM_LIVE = 'yes'
.\sender.exe session --mode calibrate --hours 0.34 --max-spend-lamports 300000 --offsets -300,-100,0,100
Remove-Item Env:CONFIRM_LIVE
.\sender.exe collect
# Expect: "starting calibrate session ..."; after 20 minutes it ends by itself; collect says "0 still pending".
# STOP and send the screen text: any "refused" message (nothing was sent), "RebootRequired" (restart Windows, then again), "platform".

=== STEP 9: the main session, 3.5 hours (start ONLY at the time agreed with the organiser; the window stays open) ===
cd C:\slwlab-b
$env:CONFIRM_LIVE = 'yes'
.\sender.exe session --mode measure --hours 3.5 --max-spend-lamports 3356350 --prereg 05a3b2338f324021b779b80ae1122b226b657fb1 --allocation allocation\alloc_platform_b.json
Remove-Item Env:CONFIRM_LIVE
# Expect: "starting measure session <id> until <UTC end>". Do not close the window, do not close the lid, do not press Win+L, do not sign out.
# If the laptop switches off or the window closes: do NOT start again; tell the organiser (a live session is never resumed).

=== STEP 10: after the session ends ===
cd C:\slwlab-b
.\sender.exe collect
.\sender.exe reconcile --since $POST
# Expect: "ok": true, "problems": [], "open_reservations": 0. (If $POST is empty after a restart of the window: use the file name you wrote down.)
.\sender.exe sweep
# Expect: "dry-run: nothing sent", about 33 transfers.
$env:CONFIRM_LIVE = 'yes'
.\sender.exe sweep --live --max-spend-lamports 200000
Remove-Item Env:CONFIRM_LIVE
.\sender.exe balances
# Expect: the total is again about 30000000 minus the fees (about 3.9 million) - the rest is back on the master.

=== STEP 11: pack the data and send it ===
cd C:\slwlab-b
Compress-Archive -Path .\data -DestinationPath .\data_B.zip -Force
Get-Item .\data_B.zip | Select-Object Name, Length
(Get-FileHash .\data_B.zip -Algorithm SHA256).Hash.ToLower()
# Send data_B.zip and the 64-character hash line to the organiser. NOT the folder secrets, NOT the file .env, NOT a screenshot of the .env.
# When the organiser confirms receipt, delete the folder C:\slwlab-b\secrets (the master is empty or almost empty) and, if you want, the Helius key in your dashboard.
